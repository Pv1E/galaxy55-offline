# ponen. Sin tutorial, las tablas se ponen otra vez. -Timeout 300 -Extra "--lang=fr"
# Esperado: tablas -> blockl 59 62 3 vacio + aviso rojo "seule la Bauxite ou l'Aluminium"; bauxita -> BMinAluminium en 59,62,3;
