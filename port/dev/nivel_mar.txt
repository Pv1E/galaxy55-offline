# bajada del nivel del mar (net/Local.hx y Game.onCommand(SReduceWater) del original), en BIWinterPeaks (mar en z = 32):
#   powershell -File port/dev/devrun.ps1 -Script port/dev/nivel_mar.txt -Extra "--bgbiome=BIWinterPeaks" -Timeout 300
# Una zanja abierta desde el lago y una cueva en un cerro de roca; con el contador a 12 bloques de la bajada, el agua que entra baja el
# mar una capa en todo el planeta. Capturas antes (1) y despues (2) de la bajada y despues de guardar y volver a cargar (3):
# el lago desde la colina, la orilla, bajo el agua y la cueva. lightcheck compara la luz con la de recalcular todo el planeta.
# Sin olas, para que las capturas 2 y 3 (la luz de la bajada y la recalculada al cargar) se puedan comparar.
# Todo va respecto a la orilla del lago mas cercana a 100,105 (find shore: fx fy = el agua, la playa en x + 1; fz = 31, el agua
# de arriba del mar)
wait 3
planet
wait 30
trace
waves 0
print sea
find shore 100 105
# cueva: un cerro de roca junto a la playa con un tunel (z 31..35) y un saliente seco en y = 106; zanja abierta en la playa
# (z 31..32) entre el lago y la boca del tunel
filla fx+7 fy-2 fx+16 fy+3 fz fz+6 BWinterRock
filla fx+7 fy fx+14 fy fz fz+4 none
filla fx+7 fy+1 fx+14 fy+1 fz+2 fz+4 none
filla fx+2 fy fx+6 fy fz fz+1 none
wait 29
tp fx+10.5 fy-4.5 top 3.1416 -0.35
wait 20
shot nm_lago1
tp fx+4.5 fy-1.5 top 3.1416 -0.45
wait 2
shot nm_orilla1
tp fx-7.5 fy+0.5 fz-2.5 0 0.35
wait 2
shot nm_agua1
tp fx+13.5 fy+1.5 fz+2.01 3.1416 -0.3
wait 2
shot nm_cueva1
print lightcheck
# se abre la zanja al lago: el mar entra en la zanja y en la cueva
seaflood 12
breakp fx+1 fy fz+1
breakp fx+1 fy fz
wait 600
print sea
print block fx+4 fy fz+1
print block fx+4 fy fz
print block fx+12 fy fz+1
print block fx+12 fy fz
print lightcheck
tp fx+10.5 fy-4.5 top 3.1416 -0.35
wait 20
shot nm_lago2
tp fx+4.5 fy-1.5 top 3.1416 -0.45
wait 2
shot nm_orilla2
tp fx-7.5 fy+0.5 fz-2.5 0 0.35
wait 2
shot nm_agua2
tp fx+13.5 fy+1.5 fz+2.01 3.1416 -0.3
wait 2
shot nm_cueva2
# el agua que fluye solo llega al nivel nuevo
breakp fx+15 fy fz
breakp fx+15 fy fz+1
wait 60
print block fx+15 fy fz
print block fx+15 fy fz+1
# guardar y volver a cargar: el nivel y el contador se guardan con la partida
closewin
reload
wait 600
print sea
print block fx+4 fy fz+1
print block fx-12 fy fz+1
tp fx+10.5 fy-4.5 top 3.1416 -0.35
wait 20
shot nm_lago3
tp fx+4.5 fy-1.5 top 3.1416 -0.45
wait 2
shot nm_orilla3
tp fx+13.5 fy+1.5 fz+2.01 3.1416 -0.3
wait 2
shot nm_cueva3
waves 1
quit
