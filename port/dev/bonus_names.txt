# nombres de los cristales de hexalita (BBonus*): esquina del bloque apuntado e inventario. Probar con -Extra "--lang=fr" (en, es)
wait 3
planet
wait 30
print names BBonusSmall BBonusMedium BBonusLarge BBonusCharge
fillt 55 55 79 73 10 10 BRock
fillt 55 55 79 73 11 20 none
setp 66 64 top BBonusSmall
setp 66 60 top BBonusMedium
setp 66 68 top BBonusLarge
slot 0
tp 64.5 64.5 top
find BBonusSmall 4
lookat fx+0.5 fy+0.5 fz+0.2
wait 9
print select
shot bn_small
tp 64.5 60.5 top
find BBonusMedium 4
lookat fx+0.5 fy+0.5 fz+0.3
wait 9
print select
shot bn_medium
tp 64.5 68.5 top
find BBonusLarge 4
lookat fx-0.1 fy+0.5 fz+0.5
wait 9
print select
shot bn_large
# inventario: el nombre sale bajo la casilla al recibir el bloque
give BBonusLarge 1
wait 5
shot bn_inv
quit
